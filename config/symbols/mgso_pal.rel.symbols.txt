# mgso_pal.rel symbols — GGSPA4 (PAL), the engine overlay
#
# Recovered by the process in docs/decompilation-process.md. The engine is
# Konami's own code and NO PUBLIC DECOMPILATION OF IT EXISTS, so none of the
# routes that named main.dol apply here: there is no reference binary to align
# against and no upstream source to match. Everything in this file was
# established by reading the instruction stream in this binary.
#
# Format: <section> <offset> <size> <name> <origin>
#
# The same five columns as main.dol.symbols.txt so one parser reads both -
# the difference is that the number is an OFFSET here, not an address.
#
# OFFSETS ARE INTO THE REL's .text, not runtime addresses. The module loads at
# 0x7F008000 and .text begins at 0x7F0080EC, so runtime = offset + 0x7F0080EC.
# Recorded as offsets because the load address is the game's choice and has
# already changed once.
#
# TWO KINDS OF NAME LIVE HERE, AND THEY ARE NOT THE SAME CLAIM.
#
#   PascalCase / SDK spelling  — the function's ACTUAL name, where the binary
#       identifies it beyond doubt. `sqrtf` is sqrtf: frsqrte followed by
#       three Newton-Raphson rounds, leaving its ARGUMENT UNTOUCHED for
#       non-positive input, is that function and no other.
#
#       An earlier revision of this note said "returning zero for non-positive
#       input", which is wrong and was never checked against the branch. The
#       `ble` path does not touch f1, so sqrtf(-3.0f) returns -3.0f, not 0.0f.
#       Zero is merely what the case x == 0 happens to produce. The error
#       mattered: it is the one detail that distinguishes this function from
#       main.dol's `sqrt_abs` at 0x80006668, which is byte-for-byte the same
#       except that it substitutes |x| first - so a reader working from the
#       wrong note would have called them the same function.
#
#   lower_snake_case           — OUR DESCRIPTION of what the function does.
#       Konami's own identifier for it is unknown and is not being guessed at.
#       `gcn_emit_sphere_strips` is not a claim that anyone ever typed those
#       words; it is a claim about what the code does, which is checkable
#       against the disassembly and is what a name is for here.
#
# Mixing the two without marking them would be dishonest about which is
# evidence and which is a label, so the case distinction carries it.
#
# ORIGINS USED IN THIS FILE
#
#   own            read out of this binary's instruction stream by us
#   message        fixed by a string the function itself stores or passes
#   closed-region  a gap bounded by two confirmed neighbours, holding exactly
#                  as many functions as the reference source has names for it,
#                  in a fixed order. Counted, not chosen; a gap that does not
#                  close exactly proposes nothing
#   struct-abi     the argument shape and the byte offsets the function reads
#                  from its parameters match a PUBLIC third-party struct
#                  layout uniquely. No reference binary and no message
#
# No symbol here may originate from leaked source. See project rule 9.

# --- CodeWarrior / libm ------------------------------------------------------
.text 0x017834 0x70 sqrtf own

# --- a shared entry pool ----------------------------------------------------
# One object serves the whole engine: every caller traced passes the same
# pool, 0x7F4EF794 at runtime, and distinguishes itself by a `kind` in r4.
#
#   gcn_pool_acquire(pool, kind)  walks from pool->0x14 to pool->0x24 looking
#       for an entry whose header word EQUALS `kind` (`cmpw r0, r28`, r28
#       being the second argument), and returns it, or 0 if there is none -
#       and 0 immediately if pool->0x34 is non-zero. A header of 0xFF is a
#       wrap marker and sends the cursor back to pool->0x8; 0 means the slot
#       is empty. Entries are 0x10 bytes of header followed by the payload,
#       which is what the caller receives: the header sits at result-0x10.
#
#       IT FINDS, IT DOES NOT ALLOCATE. The name is ours and "acquire" was
#       first read as "allocate", which made 79% of calls returning NULL look
#       like exhaustion instead of "no record of that kind is waiting". See
#       the correction in F202.
#   gcn_pool_free(pool, entry)    writes 0 to that header: the record is
#       consumed and the slot is empty again.
#   gcn_pool_clear_entry_flag(pool, entry)  clears BIT 7 of the header tag and
#       stores it back, leaving the record present with a changed tag - so a
#       later search for the original kind will not find it, but one for the
#       cleared tag will.
#
# All three take the same lock (`bss_55C74`) around the header write.
#
# `fn_1_1325D4` at 0x1325D4 is left unnamed on purpose: it returns
# `*(entry - 0xC) - 0x10` in three instructions, and while callers use it as
# a predicate, what that field MEANS is not established. A name would be a
# guess dressed as a finding.
.text 0x1323C4 0x11C gcn_pool_acquire own
.text 0x1324E0 0x054 gcn_pool_clear_entry_flag own
.text 0x132534 0x050 gcn_pool_free own

# --- the movie player, in mpegGCN.c -----------------------------------------
# `mpeg_movie_task` is the function the engine's task table holds for the
# movie - the table entry reads `fn 0x7F151214`, which is this - and the
# dispatcher calls a task with r3 still holding its own list node, so the
# node IS this function's argument. It is a state machine on node+0x44:
# 0 opens the movie by id (the two asserts at mpegGCN.c lines 905 and 910
# guard the lookup and the attach), 1 waits, 2 plays, 3 ends.
#
# State 1 leaves only when `(*bss_55EA4)->0x3C` is non-zero, and
# `mpeg_poll_stream_events` is what sets it - on an event whose code is 1,
# taken from `gcn_event_poll`. That is the stall of F196: the poll returns
# nothing at all while the task mask is non-zero.
.text 0x149048 0x0E0 mpeg_poll_stream_events own
.text 0x149128 0x5CC mpeg_movie_task own

# --- the per-frame task scheduler (see HANDOFF F94) --------------------------
# `gcn_task_table_init` lays the table out and is the ONLY `stw` to the mask
# in the whole overlay: twelve entries of stride 0x44 from bss+0x23708, each
# given its gate at +0x40 from the halfword table at data_D5E0 and its list
# head pointed at itself, then bss+0x23A38 stored as zero. Watching that word
# every step confirms it - the only two changes seen are the loader's memcpy
# bringing the image in and this function zeroing it.
#
# `gcn_event_poll` is the engine's event pump, and its first act is the one
# that matters: it reads the global task mask and returns ZERO EVENTS if it
# is non-zero, so parking the game does not merely gate task levels, it stops
# every event in the engine from being delivered (F196).
# And the two that CHANGE it. They are why a search for `stw ... @l` found
# only the initialiser: both use `lwzu` to load the mask and leave its address
# in the base register, so the store is `stw r0, 0x0(r4)` and names no label
# at all. Three instructions each, and unambiguous:
#
#     gcn_task_mask_set(bits)    mask |=  bits      /* park */
#     gcn_task_mask_clear(bits)  mask &= ~bits      /* unpark */
#
# Watching the mask every step shows exactly four calls in a boot: set 0x1,
# clear 0x1, and set 0x8 - which is the one the movie never recovers from
# (F200).
.text 0x0F45EC 0x014 gcn_task_mask_set own
.text 0x0F4600 0x014 gcn_task_mask_clear own
.text 0x0F38CC 0x080 gcn_task_table_init own
.text 0x0F52A8 0x140 gcn_event_poll own
# The engine's whole per-frame workload is reached through a function pointer
# in this table, which is why following `bl` targets out of the main loop
# never arrives at the renderer.
.text 0x0F394C 0xD4 gcn_task_run_all own
.text 0x0F3A20 0x38 gcn_task_link own

# --- the chunked file stream ------------------------------------------------
# The pair that reads a file a chunk at a time, and the evidence is in the
# instruction stream rather than inferred: `gcn_stream_issue_read` loads the
# callback argument for `DVDReadAsyncPrio` with
#
#     addi r30, r3, fn_1_320@l     ...     mr r7, r30
#
# so the function it hands the drive as its completion IS the one named
# `gcn_stream_read_done` below. Nothing about that is a guess.
#
# gcn_stream_read_done, on each completion:
#
#     if (desc->0x08 > 0)  gcn_stream_issue_read(desc);   /* next chunk */
#     else { ...; if (desc->0x0C) (*desc->0x0C)(ctx, status); }
#
# so 0x08 is the count still to read and 0x0C the stream's OWN completion,
# called once the last chunk lands. `desc->0x20` is an embedded DVDFileInfo -
# it is what is passed as the file - and `desc->0x9C` points at the context
# holding the running offset and length.
#
# gcn_stream_issue_read RETRIES: `DVDReadAsyncPrio` returning 0 branches back
# and asks again, with priority 2. Worth knowing, because a drive model that
# refused reads would spin here rather than fail.
#
# THIS PATH IS NOT THE MOVIE FAULT (F195). It ran 406 reads with 406
# callbacks and no refusals, and the movie's own 8 chunks completed through
# it correctly.
.text 0x000320 0x130 gcn_stream_read_done own
.text 0x000450 0x0F8 gcn_stream_issue_read own

# --- a service thread and its mailbox ---------------------------------------
# Found by dumping the blocked guest threads: `gcn_worker_loop` is the
# OUTERMOST frame on a thread's stack, so it is that thread's body, and
# nothing above it exists to be named. The three share one object, whose
# fields they fix between them:
#
#     obj +0x8020  status, written by the worker (0 idle, 0x40000 busy)
#     obj +0x8024  the request, written by whoever asks
#     obj +0x8068  the request latched by the worker as it takes it
#     obj +0x8D94  the OSMessageQueue it waits on - one slot, a handoff
#
# `gcn_worker_loop` is a state machine: state 0 marks itself idle, blocks for
# a request, marks itself busy and takes the next state from the message;
# state 1 does the work; states 2 and >=4 reset to 0.
#
# STATE 3 BRANCHES BACK WITH THE STATE UNCHANGED - an infinite loop that
# never yields and never sleeps. Nothing has been seen to reach it, and it is
# recorded because a thread that reached it would be indistinguishable from a
# hang and would have no stack to explain itself.
.text 0x000264 0x4C gcn_worker_take_request own
.text 0x0002B0 0x0C gcn_worker_set_status own
.text 0x000804 0x84 gcn_worker_loop own

# --- sphere-map geometry (gcn_dgd.c / gcn_spheremap.c) -----------------------
# 32 strips of 66 vertices, position and normal per vertex, built from a
# GXBegin(GX_TRIANGLESTRIP, GX_VTXFMT2, 66) and two three-float writes to the
# write-gather pipe per vertex. The z of each normal comes from sqrtf of
# 1 - (x*x + y*y), which is what makes it a unit sphere.
.text 0x12106C 0x20C gcn_emit_sphere_strips own
.text 0x121278 0x4 gcn_gx_end own
.text 0x12127C 0x14 gcn_gx_write3f own
.text 0x121290 0x14 gcn_gx_write3f_pos own

# --- zlib, embedded in the engine ------------------------------------------
# PascalCase rule aside, these are zlib's OWN names: the binary identifies
# them beyond doubt, so they are the actual names and not a description.
#
# The evidence is the five error messages `inflate_trees_dynamic` stores into
# `z->msg`, which appear in this binary verbatim and in that order:
#
#   "oversubscribed literal/length tree"   "incomplete literal/length tree"
#   "oversubscribed distance tree"         "incomplete distance tree"
#   "empty distance tree with lengths"
#
# and the structure agrees independently: a 16-entry bit-length histogram
# (zlib's BMAX is 15), the literal/length count 0x101 = 257, the table-fill
# stride `1 << (k - w)`, and the return codes -3, -4 and -5 - Z_DATA_ERROR,
# Z_MEM_ERROR and Z_BUF_ERROR. Two routes, one answer.
#
# WHY THIS MATTERS BEYOND THE NAMES. huft_build holds 81.6% of the boot's
# samples (HANDOFF F132). The thing the boot spends its life doing is
# building Huffman tables, which is not where anyone was looking.
#
# THREE MORE, BY A CLOSED REGION AND BY A STRUCT LAYOUT.
#
# inftrees.c defines exactly four functions, in this order: huft_build,
# inflate_trees_bits, inflate_trees_dynamic, inflate_trees_fixed. Two of them
# are already fixed above by their error messages, and in the binary the four
# slots are EXACTLY CONTIGUOUS - every gap is zero bytes:
#
#     0x0F0F2C +0x4A4  huft_build              (message)
#     0x0F13D0 +0x0A0  ...one slot, one name   -> inflate_trees_bits
#     0x0F1470 +0x17C  inflate_trees_dynamic   (message)
#     0x0F15EC +0x038  ...one slot, one name   -> inflate_trees_fixed
#     0x0F1624 +0x180  a different translation unit
#
# Two anchors, two gaps, two remaining functions in a fixed source order:
# nothing is chosen here, it is counted. 0x38 for inflate_trees_fixed also
# agrees with the build: without BUILDFIXED its whole body is four
# assignments and a return.
#
# inflate_fast is identified differently, by the shape of its parameters.
# zlib 1.1.x declares it
#
#     inflate_fast(uInt bl, uInt bd, inflate_huft *tl, inflate_huft *td,
#                  inflate_blocks_state *s, z_streamp z)
#
# so r7 and r8 are struct pointers, and its first five lines read z->next_in,
# z->avail_in, s->bitb, s->bitk, s->write and s->read, then compute
# m = q < s->read ? s->read - q - 1 : s->end - q. The prologue at 0x0F02F0
# does exactly that, in that order, and the offsets it uses are
#
#     r8 +0x00 +0x04                    next_in, avail_in   (z_stream)
#     r7 +0x1C +0x20 +0x24              bitk, bitb, hufts
#     r7 +0x28 +0x2C +0x30 +0x34        window, end, read, write
#
# which is inflate_blocks_state's published layout, field for field. Seven
# offsets agreeing by chance is not a thing that happens. The body then holds
# the LZ77 window copy, two byte loops with the window base reloaded between
# them for the wrap - and it is the hottest function in the whole run, which
# is what inflate_fast exists to be. This is a 1.1.x zlib, not 1.2.x, whose
# inflate_fast takes two arguments and could not produce this prologue.
.text 0x0F02F0 0x454 inflate_fast struct-abi
.text 0x0F0F2C 0x4A4 huft_build message
.text 0x0F13D0 0x0A0 inflate_trees_bits closed-region
.text 0x0F1470 0x17C inflate_trees_dynamic message
.text 0x0F15EC 0x038 inflate_trees_fixed closed-region
.text 0x28D9E4 0x26C NewFallingFloor message

# ---- the streamed-media pipeline (F228-F232) -------------------------------
#
# Named by reading each function's whole instruction stream while tracing the
# movie stall, and each one is load-bearing in that chain. The record pool is
# a ring of {tag, size, ...} records with a payload at +0x10: `gcn_pool_pump`
# advances the read cursor past consumed (tag 0) records and stops at the
# first live one, and refills only when more than a third of the buffer is
# free, which is why a pinned head starves the ring rather than growing it.
# Tag 0xFF is the wrap marker and tag 0xF0 the end-of-stream marker.
#
# `fn_1_9C8` at 0x9C8 is deliberately NOT named here. It returns the field at
# +0x8020 of the global at bss_3020 - the same field `gcn_worker_set_status`
# (0x2B0) writes - and `gcn_pool_pump` refuses to refill while it is
# non-zero. Watching it through a run shows it taking 0x80000, 0x40000 and 0,
# which reads as a byte count rather than a status, so the established name
# and the observed values do not sit comfortably together. Naming it either
# way would resolve that disagreement by assertion instead of by evidence.
